//HALF ADDER 
 
module half_adder_gate(sum, carry, a, b); input a, b; 
output sum, carry; 
 
xor(sum, a, b); 
and(carry, a, b); 
 
endmodule 
 
 
 
module half_adder_dataflow(sum, carry, a, b); 
 
input a, b; output sum, carry; 
 
assign sum = a ^ b; 
assign carry = a & b; 
 
endmodule

module half_adder(sum, carry, a, b); 
 
input a, b; output sum, carry; xor(sum, a, b); and(carry, a, b); with dataflow //assign sum=a^b; 
//assign carry=a&b; 
 
endmodule 

module full_adder(sum, carry, a, b, cin); 
 
input a, b, cin; 
output sum, carry; 
 
wire s1; wire c1; wire c2; 
 
 
 
half_adder HA1(s1, c1, a, b); half_adder HA2(sum, c2, s1, cin); 
or(carry, c1, c2); //assign carry=c1|c2; 
 
endmodule
