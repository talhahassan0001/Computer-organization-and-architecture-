
module pc_reg (
    input             clk, reset,
    input      [31:0] next_pc,
    output reg [31:0] pc
);
    
    initial pc = 32'h0000_0000;

    always @(posedge clk) begin
        if (reset) pc <= 32'h0000_0000;
        else       pc <= next_pc;
    end
endmodule


module pc_adder4 (
    input  [31:0] pc,
    output [31:0] pc4
);
    assign pc4 = pc + 32'd4;
endmodule


module mux2 (
    input  [31:0] d0, d1,
    input         sel,
    output [31:0] y
);
    assign y = sel ? d1 : d0;
endmodule


module mux4 (
    input      [31:0] d0, d1, d2, d3,
    input      [1:0]  sel,
    output reg [31:0] y
);
    always @(*) begin
        case (sel)
            2'b00:   y = d0;
            2'b01:   y = d1;
            2'b10:   y = d2;
            default: y = d3;
        endcase
    end
endmodule


module imem (
    input  [31:0] pc,
    output [31:0] inst
);
    reg [31:0] mem [0:63];
    integer i;
    initial begin
        for (i = 0; i < 64; i = i + 1)
            mem[i] = 32'h0000_0013;         

    end


    assign inst = (pc[31:8] == 24'd0) ? mem[pc[7:2]] : 32'h0000_0013;
endmodule


module regfile (
    input         clk, RegWEn,
    input  [4:0]  AddrA, AddrB, AddrD,
    input  [31:0] DataD,
    output [31:0] DataA, DataB
);
    reg [31:0] regs [0:31];
    integer i;
    initial begin
        for (i = 0; i < 32; i = i + 1)
            regs[i] = 32'd0;
    end


    assign DataA = (AddrA == 5'd0) ? 32'd0 : regs[AddrA];
    assign DataB = (AddrB == 5'd0) ? 32'd0 : regs[AddrB];

    

    always @(posedge clk) begin
        if (RegWEn && (AddrD != 5'd0))
            regs[AddrD] <= DataD;
    end
endmodule

module immgen (
    input      [31:0] inst,
    input      [1:0]  ImmSel,
    output reg [31:0] Imm
);
    always @(*) begin
        case (ImmSel)
            2'b00: Imm = {{20{inst[31]}}, inst[31:20]};                                      
            2'b01: Imm = {{20{inst[31]}}, inst[31:25], inst[11:7]};                          
            2'b10: Imm = {{19{inst[31]}}, inst[31], inst[7], inst[30:25], inst[11:8], 1'b0};   
            2'b11: Imm = {{11{inst[31]}}, inst[31], inst[19:12], inst[20], inst[30:21], 1'b0}; 
            default: Imm = 32'd0;
        endcase
    end
endmodule


module branch_comp (
    input  [31:0] DataA, DataB,
    output        BrEq, BrLT
);
    assign BrEq = (DataA == DataB);
    assign BrLT = ($signed(DataA) < $signed(DataB));
endmodule


module alu (
    input      [31:0] A, B,
    input      [2:0]  ALUSel,
    output reg [31:0] result
);
    always @(*) begin
        case (ALUSel)
            3'b000:  result = A + B;
            3'b001:  result = A - B;
            3'b010:  result = A & B;
            3'b011:  result = A | B;
            3'b100:  result = A ^ B;
            3'b101:  result = ($signed(A) < $signed(B)) ? 32'd1 : 32'd0;
            default: result = 32'd0;
        endcase
    end
endmodule


module datapath_top (
    input         clk, reset,
    input         PCSel, RegWEn, ASel, BSel,
    input  [2:0]  ALUSel,
    input  [1:0]  ImmSel,
    input  [1:0]  WBSel,
    output [31:0] pc, inst, alu_result, wb,
    output        BrEq, BrLT
);
    wire [31:0] pc4, next_pc, DataA, DataB, Imm, alu_a, alu_b;

    pc_adder4 ADD4  (.pc(pc), .pc4(pc4));
    mux2      PCMUX (.d0(pc4), .d1(alu_result), .sel(PCSel), .y(next_pc));
    pc_reg    PCR   (.clk(clk), .reset(reset), .next_pc(next_pc), .pc(pc));


    imem      IM    (.pc(pc), .inst(inst));

   
    regfile   RF    (.clk(clk), .RegWEn(RegWEn),
                     .AddrA(inst[19:15]), .AddrB(inst[24:20]), .AddrD(inst[11:7]),
                     .DataD(wb), .DataA(DataA), .DataB(DataB));


    immgen      IG  (.inst(inst), .ImmSel(ImmSel), .Imm(Imm));
    branch_comp BC  (.DataA(DataA), .DataB(DataB), .BrEq(BrEq), .BrLT(BrLT));


    mux2      AMUX  (.d0(DataA), .d1(pc),  .sel(ASel), .y(alu_a));
    mux2      BMUX  (.d0(DataB), .d1(Imm), .sel(BSel), .y(alu_b));
    alu       ALU0  (.A(alu_a), .B(alu_b), .ALUSel(ALUSel), .result(alu_result));

    mux4      WBMUX (.d0(alu_result), .d1(pc4), .d2(32'd0), .d3(32'd0),
                     .sel(WBSel), .y(wb));
endmodule
`timescale 1ns/1ps

module tb_datapath;
    reg clk = 0, reset = 1;
    wire PCSel, RegWEn, ASel, BSel;
    wire [2:0] ALUSel;
    wire [1:0] ImmSel, WBSel;
    wire [31:0] pc, inst, alu_result, wb;
    wire BrEq, BrLT;

    integer errors = 0;

    datapath_top dut (
        .clk(clk), .reset(reset),
        .PCSel(PCSel), .RegWEn(RegWEn), .ASel(ASel), .BSel(BSel),
        .ALUSel(ALUSel), .ImmSel(ImmSel), .WBSel(WBSel),
        .pc(pc), .inst(inst), .alu_result(alu_result), .wb(wb),
        .BrEq(BrEq), .BrLT(BrLT)
    );

    
    always #5 clk = ~clk;

 
    function [10:0] ctrl(input [5:0] idx, input beq);
        case (idx)
            0, 1:    ctrl = {1'b0, 1'b1, 1'b0, 1'b1, 3'b000, 2'b00, 2'b00}; 
            2:       ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b000, 2'b00, 2'b00};
            3:       ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b001, 2'b00, 2'b00}; 
            4:       ctrl = {beq,  1'b0, 1'b1, 1'b1, 3'b000, 2'b10, 2'b00}; 
            5:       ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b000, 2'b00, 2'b00};
            6:       ctrl = {beq,  1'b0, 1'b1, 1'b1, 3'b000, 2'b10, 2'b00};
            7, 8:    ctrl = {1'b0, 1'b1, 1'b0, 1'b1, 3'b000, 2'b00, 2'b00}; 
            9:       ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b101, 2'b00, 2'b00}; 
            10:      ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b010, 2'b00, 2'b00};
            11:      ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b011, 2'b00, 2'b00}; 
            12:      ctrl = {1'b0, 1'b1, 1'b0, 1'b0, 3'b100, 2'b00, 2'b00}; 
            13:      ctrl = {1'b0, 1'b1, 1'b0, 1'b1, 3'b000, 2'b00, 2'b00}; 
            default: ctrl = 11'b0;                                          
        endcase
    endfunction

    assign {PCSel, RegWEn, ASel, BSel, ALUSel, ImmSel, WBSel} = ctrl(pc[7:2], BrEq);

 
    task check_reg(input integer idx, input [31:0] expected);
        begin
            if (dut.RF.regs[idx] === expected)
                $display("  PASS: x%0d = %0d (0x%08h)", idx, $signed(dut.RF.regs[idx]), dut.RF.regs[idx]);
            else begin
                $display("  FAIL: x%0d = %0d, expected %0d", idx, $signed(dut.RF.regs[idx]), $signed(expected));
                errors = errors + 1;
            end
        end
    endtask

    task check_val(input [255:0] name, input [31:0] actual, input [31:0] expected);
        begin
            if (actual === expected)
                $display("  PASS: %0s = %0d", name, $signed(actual));
            else begin
                $display("  FAIL: %0s = %0d, expected %0d", name, $signed(actual), $signed(expected));
                errors = errors + 1;
            end
        end
    endtask

  
    always @(posedge clk) begin
        if (!reset)
            $display("t=%0t pc=%0d inst=%h A+B=%0d wb=%h RegWEn=%b PCSel=%b BrEq=%b BrLT=%b",
                     $time, pc, inst, alu_result, wb, RegWEn, PCSel, BrEq, BrLT);
    end

    initial begin
        $dumpfile("datapath.vcd");
        $dumpvars(0, tb_datapath);

  
        dut.IM.mem[0]  = 32'h00700293; 
        dut.IM.mem[1]  = 32'h00500313; 
        dut.IM.mem[2]  = 32'h006283B3; 
        dut.IM.mem[3]  = 32'h40628433; 
        dut.IM.mem[4]  = 32'h00038463; 
        dut.IM.mem[5]  = 32'h008384B3; 
        dut.IM.mem[6]  = 32'h00000463; 
        dut.IM.mem[7]  = 32'h06300513;
        dut.IM.mem[8]  = 32'hFFD00593; 
        dut.IM.mem[9]  = 32'h0055A633; 
        dut.IM.mem[10] = 32'h0053F6B3; 
        dut.IM.mem[11] = 32'h0053E733; 
        dut.IM.mem[12] = 32'h0053C7B3;
        dut.IM.mem[13] = 32'h00500013; 

       
        $display("--- Test: Reset ---");
        #12;                         
        check_val("pc after reset", pc, 32'd0);
        reset = 0;

    
        $display("--- Running program ---");
        repeat (13) @(posedge clk);
        #1;

        
        $display("--- Register results ---");
        check_reg(5,  32'd7);
        check_reg(6,  32'd5);
        check_reg(7,  32'd12);
        check_reg(8,  32'd2);
        check_reg(9,  32'd14);
        check_reg(10, 32'd0);         
        check_reg(11, 32'hFFFFFFFD);
